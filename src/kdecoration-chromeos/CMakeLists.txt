# [CHANGE: claude-code | 2026-10-10] Luminos ChromeOS-style KWin decoration (DECISION 135)
cmake_minimum_required(VERSION 3.20)
project(luminos-kdecoration-chromeos VERSION 1.0.0 LANGUAGES CXX)

find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(QT_MAJOR_VERSION 6)
include(KDEInstallDirs6)
include(KDECMakeSettings)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(KF6 REQUIRED COMPONENTS CoreAddons Config)
find_package(KDecoration3 REQUIRED)
add_library(luminoschromeos MODULE chromeosdecoration.cpp chromeosbutton.cpp chromeosshadow.cpp)
set_target_properties(luminoschromeos PROPERTIES OUTPUT_NAME org.kde.luminos.chromeos)
target_link_libraries(luminoschromeos PRIVATE Qt6::Gui KF6::CoreAddons KF6::ConfigCore KDecoration3::KDecoration)
install(TARGETS luminoschromeos DESTINATION ${KDE_INSTALL_PLUGINDIR}/${KDECORATION_PLUGIN_DIR})
